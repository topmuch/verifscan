   344→  const expired = isExpired(lot.expirationDate);
   345→  const remainingDays = daysUntil(lot.expirationDate);

   347→  const allergens = detectAllergens(lot.ingredients);
   348→  const currentUrl = typeof window !== "undefined" ? window.location.href : "";

   368→              size="sm"
   369→              onClick={toggle}
   370→              title={theme === "dark" ? "Mode clair" : "Mode sombre"}

   399→            <div className="mx-auto max-w-5xl px-4 sm:px-6 lg:px-8 py-3 flex flex-wrap items-center gap-2">
   400→              <span className="text-xs font-medium text-gray-500 dark:text-gray-400 mr-2">Partager :</span>
   401→              <a
   402→                href={`https://wa.me/?text=${encodeURIComponent(shareText)}&url=${encodeURIComponent(currentUrl)}`}
   403→                target="_blank"
   404→                rel="noopener noreferrer"
   405→                className="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-lg bg-[#25D366] text-white text-xs font-medium hover:opacity-90"
   406→              >
   407→                <MessageCircle className="size-3.5" />
   408→                WhatsApp
   409→              </a>
   410→              <a
   411→                href={`https://www.facebook.com/sharer/sharer.php?u=${encodeURIComponent(currentUrl)}`}
   412→                target="_blank"
   413→                rel="noopener noreferrer"
   414→                className="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-lg bg-[#1877F2] text-white text-xs font-medium hover:opacity-90"
   415→              >
   416→                <svg className="size-3.5" fill="currentColor" viewBox="0 0 24 24">
   417→                  <path d="M24 12.073c0-6.627-5.373-12-12-12s-12 5.373-12 12c0 5.99 4.388 10.954 10.125 11.854v-8.385H7.078v-3.47h3.047V9.43c0-3.007 1.792-4.669 4.533-4.669 1.312 0 2.686.235 2.686.235v2.953H15.83c-1.491 0-1.956.925-1.956 1.874v2.25h3.328l-.532 3.47h-2.796v8.385C19.612 23.027 24 18.062 24 12.073z" />
   418→                </svg>
   419→                Facebook

  1300→function Reveal({ children }: { children: React.ReactNode }) {
  1301→  const ref = useRef<HTMLDivElement>(null);
  1302→  const [visible, setVisible] = useState(false);
  1303→
  1304→  useEffect(() => {
  1305→    const el = ref.current;
  1306→    if (!el) return;
  1307→    const obs = new IntersectionObserver(
  1308→      (entries) => {
  1309→        entries.forEach((e) => {
  1310→          if (e.isIntersecting) {
  1311→            setVisible(true);
  1312→            obs.unobserve(e.target);
  1313→          }
  1314→        });
  1315→      },
  1316→      { threshold: 0.08, rootMargin: "0px 0px -40px 0px" }
  1317→    );
  1318→    obs.observe(el);
  1319→    return () => obs.disconnect();
  1320→  }, []);
  1321→
  1322→  return (
  1323→    <div ref={ref} className={cn("vs-section-reveal", visible && "vs-revealed")}>
  1324→      {children}
  1325→    </div>
  1326→  );
  1327→}

  1513→      </div>

  1516→}

  1518→/* ============ Review modal ============ */
  1519→
  1520→/**
  1521→ * Modal for a consumer to leave a review on a product. Submitting POSTs to
  1522→ * /api/lots/[id]/reviews — the merchant is notified by email + in-app
  1523→ * notification on the server side. No auth required.
  1524→ *
  1525→ * After a successful submit, the parent's `onSubmitted` callback is invoked

  1528→function ReviewModal({

  1537→  onClose: () => void;

  1546→  const [reviewerName, setReviewerName] = useState("");

  1570→    window.addEventListener("keydown", onKey);
  1571→    return () => window.removeEventListener("keydown", onKey);

  1597→          rating,
  1598→          comment: comment.trim() || null,
  1599→          reviewerName: reviewerName.trim() || null,
  1600→          reviewerPhone: reviewerPhone.trim() || null,
  1601→        }),
  1602→      });

  1604→        const data = await res.json().catch(() => ({}));

  1610→      onSubmitted(data.review);

  1613→        onClose();
  1614→      }, 1800);
  1615→    } catch (err: any) {

  1617→      setSubmitting(false);
  1618→    }
  1619→  };
  1620→
  1621→  const displayRating = hoverRating || rating;
  1622→
  1623→  return (
  1624→    <div
  1625→      className="fixed inset-0 z-[100] flex items-end sm:items-center justify-center p-0 sm:p-4"

  1634→      />
  1635→

  1637→      <div
  1638→        className="relative bg-white dark:bg-gray-900 rounded-t-2xl sm:rounded-2xl shadow-2xl w-full sm:max-w-md max-h-[92vh] overflow-y-auto"
  1639→        style={{ borderTop: `4px solid ${GREEN}` }}
  1640→      >
  1641→        {/* Header */}
  1642→        <div
  1643→          className="sticky top-0 z-10 px-5 py-4 flex items-start justify-between gap-3 border-b border-gray-100 dark:border-gray-800 bg-white dark:bg-gray-900"
  1644→        >
  1645→          <div className="min-w-0">
  1646→            <h3 id="review-modal-title" className="font-display text-lg font-bold text-gray-900 dark:text-gray-100 flex items-center gap-2">
  1647→              <Star className="size-5" style={{ color: GREEN }} />
  1648→              Donner votre avis
  1649→            </h3>
  1650→            <p className="mt-0.5 text-xs text-gray-500 dark:text-gray-400 truncate">
  1651→              {productName} <span className="opacity-70">· {brand}</span>
  1652→            </p>
  1653→          </div>
  1654→          <button
  1655→            onClick={() => !submitting && !done && onClose()}
  1656→            className="flex-shrink-0 p-1.5 rounded-lg text-gray-400 hover:text-gray-600 dark:hover:text-gray-200 hover:bg-gray-100 dark:hover:bg-gray-800"

  1683→                Votre note <span style={{ color: "#DC2626" }}>*</span>
