  1115→function Reveal({ children }: { children: React.ReactNode }) {
  1116→  const ref = useRef<HTMLDivElement>(null);
  1117→  const [visible, setVisible] = useState(false);
  1118→
  1119→  useEffect(() => {
  1120→    const el = ref.current;
  1121→    if (!el) return;
  1122→    const obs = new IntersectionObserver(
  1123→      (entries) => {
  1124→        entries.forEach((e) => {
  1125→          if (e.isIntersecting) {
  1126→            setVisible(true);
  1127→            obs.unobserve(e.target);
  1128→          }
  1129→        });
  1130→      },
  1131→      { threshold: 0.08, rootMargin: "0px 0px -40px 0px" }
  1132→    );
  1133→    obs.observe(el);
  1134→    return () => obs.disconnect();
  1135→  }, []);
  1136→
  1137→  return (
  1138→    <div ref={ref} className={cn("vs-section-reveal", visible && "vs-revealed")}>
  1139→      {children}
  1140→    </div>
  1141→  );
  1142→}
