  1037→/** Reveal wrapper — fades children in when they enter the viewport. */
  1038→function Reveal({ children }: { children: React.ReactNode }) {
  1039→  const ref = useRef<HTMLDivElement>(null);
  1040→  const [visible, setVisible] = useState(false);
  1041→
  1042→  useEffect(() => {
  1043→    const el = ref.current;
  1044→    if (!el) return;
  1045→    const obs = new IntersectionObserver(
  1046→      (entries) => {
  1047→        entries.forEach((e) => {
  1048→          if (e.isIntersecting) {
  1049→            setVisible(true);
  1050→            obs.unobserve(e.target);
  1051→          }
  1052→        });
  1053→      },
  1054→      { threshold: 0.08, rootMargin: "0px 0px -40px 0px" }
  1055→    );
  1056→    obs.observe(el);
  1057→    return () => obs.disconnect();
  1058→  }, []);
  1059→
  1060→  return (
  1061→    <div ref={ref} className={cn("vs-section-reveal", visible && "vs-revealed")}>
  1062→      {children}
  1063→    </div>
  1064→  );
  1065→}
